    // Upload
    function handleFileSelect(fileList) {
        if(!fileList || fileList.length === 0) return;
        
        const formData = new FormData();
        Array.from(fileList).forEach(file => formData.append('files', file));
        
        const progressContainer = document.getElementById('upload-progress-container');
        const bar = document.getElementById('upload-bar');
        const status = document.getElementById('upload-status');
        const percent = document.getElementById('upload-percent');
        
        progressContainer.style.display = 'block';
        status.innerText = `Uploading ${fileList.length} file(s)...`;
        
        fetch('/api/upload', {
            method: 'POST',
            body: formData,
            headers: {
                'X-File-Count': fileList.length
            }
        })
        .then(response => {
            if(!response.ok) throw new Error('Upload failed');
            // Simulate progress for better UX if server doesn't stream
            // In a real scenario, we'd use XMLHttpRequest for progress events
            return response.json();
        })
        .then(data => {
            progressContainer.style.display = 'none';
            bar.style.width = '100%';
            status.innerText = 'Upload complete';
            fetchFiles();
            fetchStorage();
            showToast(`Successfully uploaded ${data.length} files`, 'success');
        })
        .catch(err => {
            progressContainer.style.display = 'none';
            showToast('Upload failed: ' + err.message, 'error');
        });
    }

        menu.style.left = `${e.clientX}px`;
        menu.style.top = `${e.clientY}px`;

        if(rect.right > window.innerWidth) menu.style.left = `${e.clientX - rect.width}px`;
        if(rect.bottom > window.innerHeight) menu.style.top = `${e.clientY - rect.height}px`;
    }

    function contextAction(action) {
        if(!contextFile) return;
        
        if(action === 'delete') {
            if(!confirm(`Delete ${contextFile}?`)) return;
            fetch(`/api/files/${encodeURIComponent(contextFile)}`, { method: 'DELETE' })
                .then(() => {
                    fetchFiles();
                    fetchStorage();
                    showToast('File deleted', 'success');
                })
                .catch(err => showToast('Delete failed', 'error'));
        }
        else if(action === 'rename') {
            document.getElementById('renameInput').value = contextFile;
            document.getElementById('renameModal').classList.add('active');
        }
        else if(action === 'download') {
            window.location.href = `/uploads/${contextFile}`;
        }
        else if(action === 'preview') {
            const f = files.find(f => f.name === contextFile);
            if(f && imageExts.includes(f.ext)) openLightbox(contextFile);
            else if(f && videoExts.includes(f.ext)) openVideoPlayer(contextFile);
            else showToast('Preview not available for this type', 'info');
        }
        else if(action === 'copyLink') {
            const url = `${window.location.origin}/uploads/${contextFile}`;
            navigator.clipboard.writeText(url);
            showToast('Link copied to clipboard', 'success');
        }
        
        document.getElementById('contextMenu').classList.remove('active');
    }

    function saveRename() {

            method: 'PUT',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ newName })
        })
        .then(() => {
            closeModal('renameModal');
            fetchFiles();
            showToast('File renamed', 'success');
        })
        .catch(err => showToast('Rename failed', 'error'));
    }

    // Lightbox
    function openLightbox(filename) {
        const f = files.find(f => f.name === filename);
        if(!f) return;
        
        document.getElementById('lightbox-img').src = `/uploads/${filename}?t=${Date.now()}`;
        document.getElementById('lightbox-name').innerText = f.name;
        document.getElementById('lightbox-meta').innerText = `${formatSize(f.size)} • ${formatDate(f.mtime)}`;
        document.getElementById('lightbox').classList.add('active');
    }

    function closeLightbox(e) {
        if(e && e.target !== e.currentTarget && !e.target.closest('.lightbox-close')) return;
        document.getElementById('lightbox').classList.remove('active');
    }

        overlay.appendChild(vid);

    }

    // Utilities
    function formatSize(bytes) {
        if(bytes === 0) return '0 B';
        const k = 1024;
        const sizes = ['B', 'KB', 'MB', 'GB', 'TB'];
        const i = Math.floor(Math.log(bytes) / Math.log(k));
        return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
    }

    function formatDate(dateStr) {
        const d = new Date(dateStr);
        return d.toLocaleDateString() + ' ' + d.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
    }

    function getTypeIcon(ext) {
        if(imageExts.includes(ext)) return icons.image;
        if(videoExts.includes(ext)) return icons.video;
        return icons.default;
    }

    function getTypeColor(ext) {
        if(imageExts.includes(ext)) return '#22c55e';
        if(videoExts.includes(ext)) return '#f59e0b';
        return '#a1a1aa';
    }

    function setupDragAndDrop() {
        const dropzones = [document.getElementById('dropzone'), document.getElementById('dropzone-upload')];
        
        dropzones.forEach(zone => {
            if(!zone) return;
            zone.addEventListener('click', () => zone.querySelector('input').click());
            zone.addEventListener('dragover', (e) => {
                e.preventDefault();
                zone.classList.add('dragover');
            });
            zone.addEventListener('dragleave', () => {
                zone.classList.remove('dragover');
            });
            zone.addEventListener('drop', (e) => {
                e.preventDefault();
                zone.classList.remove('dragover');
                handleFileSelect(e.dataTransfer.files);
            });
        });
    }

    function closeModal(id) {
        document.getElementById(id).classList.remove('active');
    }

    function showToast(msg, type = 'info') {
        const container = document.getElementById('toastContainer');
        const toast = document.createElement('div');
        toast.className = `toast ${type}`;
        toast.innerHTML = `
            <div class="toast-icon" style="color:${type === 'success' ? 'var(--success)' : type === 'error' ? 'var(--danger)' : 'var(--accent)'}">
                <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M5 13l4 4L19 7"/></svg>
            </div>
            <div class="toast-msg">${msg}</div>
        `;
        container.appendChild(toast);
        setTimeout(() => toast.classList.add('show'), 10);
        setTimeout(() => {
            toast.classList.remove('show');
            setTimeout(() => toast.remove(), 300);
        }, 3000);
    }
